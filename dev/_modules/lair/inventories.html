"""
Gridded emission inventories.

An emission inventory is an accounting of the amount of a pollutant released
into the atmosphere, by source sector, over an area and a period, usually a
year. :class:`Inventory` holds one on a grid, one variable per sector, as a
flux with pint units, and gives its totals (:attr:`~Inventory.total_emissions`,
:attr:`~Inventory.absolute_emissions`, :meth:`~Inventory.integrate`), units
(:meth:`~Inventory.convert_units`) and grid (``clip``, ``regrid``,
``resample``). Its subclasses read the archives of EDGAR, the EPA, GFEI, Vulcan
and WetCHARTs from ``$LAIR_INVENTORY_DIR``. :doc:`/examples/inventory` totals
EDGAR over a region. The module needs the ``geo`` and ``science`` extras, and
regridding needs ESMF as well (see :ref:`optional-dependencies`).
"""

def _keep_attrs(func: Callable[_P, _R]) -> Callable[_P, _R]:
    """
    Run ``func`` with xarray keeping attributes through its operations.

    Inventory variables carry ``units``, ``long_name`` etc. that must survive
    arithmetic and reductions (older xarray drops them by default). The option
    is set only while ``func`` runs, never globally: user code keeps xarray's
    own default (GitHub issue #38).
    """

    @functools.wraps(func)
    def wrapper(*args: _P.args, **kwargs: _P.kwargs) -> _R:
        with xr.set_options(keep_attrs=True):
            return func(*args, **kwargs)

    return wrapper

_MASS_BASIS = {"NOX": "NO2"}  # NOx is conventionally reported as NO2 mass

_STEP_UNITS = {
    "annual": "year",
    "monthly": "month",
    "biweekly": "fortnight",
    "daily": "day",

def molecular_weight(pollutant: str) -> pint.Quantity:
    """
    Calculate the molecular weight of a pollutant.

    NOx is taken as NO2, the usual reporting basis for NOx emissions.

    Parameters
    ----------
    pollutant : str
        The pollutant, as a chemical formula (case-sensitive, e.g. ``'CH4'``).

    Returns
    -------
    pint.Quantity
        The molecular weight.

    See Also
    --------
    convert_units : Converts between an amount (mol) and a mass with it.

    Examples
    --------
    >>> from lair.inventories import molecular_weight
    >>> mw = molecular_weight("CH4")
    >>> f"{mw:.3f~P}"
    '16.043 g/mol'
    >>> f"{molecular_weight('NOx'):.3f~P}"  # as NO2
    '46.006 g/mol'
    """
    formula = _MASS_BASIS.get(pollutant.upper(), pollutant)
    return Formula(formula).mass * units("g/mol")

def convert_units(data: _XarrayT, pollutant: str, dst_units: Any) -> _XarrayT:
    """
    Convert the units of a data array or dataset.

    The data either has pint units (``.pint.quantify()``, what an
    :class:`Inventory` holds) or is plain, with a ``units`` attribute. It comes
    back the same way: pint units in, pint units out; plain in, plain out, with
    the ``units`` attribute and the values converted and everything else (name,
    coordinates and their attributes, other attributes) as it was. A conversion
    between an amount (mol) and a mass (kg) uses the molecular weight of
    ``pollutant``, through the ``mass_flux`` pint context.

    The data has no time steps here, so a flux per year is per pint's year of
    365.25 days, and a flux per month per a twelfth of it.
    :meth:`Inventory.convert_units` gives a flux per year on annual data the
    length of each year instead (366 days in a leap year), and a flux per month
    on monthly data the length of each month.

    Parameters
    ----------
    data : xr.DataArray | xr.Dataset
        The data to convert. A dataset converts every variable, so each needs
        units.
    pollutant : str
        The pollutant, as a chemical formula (see :func:`molecular_weight`).
    dst_units : Any
        The destination units.

    Returns
    -------
    xr.DataArray | xr.Dataset
        The data with converted units. ``data`` itself is not changed.

    Raises
    ------
    TypeError
        If ``data`` is not an xarray DataArray or Dataset.
    ValueError
        If plain data has no ``units`` attribute.

    See Also
    --------
    Inventory.convert_units : The same for an inventory's data, with the
        calendar length of each time step.
    molecular_weight : The molecular weight the conversion uses.

    Examples
    --------
    A grid of methane fluxes read from a file, in nmol m-2 s-1, to kg km-2 d-1:

    >>> import xarray as xr
    >>> from lair.inventories import convert_units
    >>> flux = xr.DataArray(
    ...     [[10.0, 20.0], [30.0, 40.0]],
    ...     dims=("y", "x"),
    ...     attrs={"units": "nmol m-2 s-1"},
    ... )
    >>> kg = convert_units(flux, "CH4", "kg km-2 d-1")
    >>> kg.attrs["units"]
    'kg km-2 d-1'
    >>> round(float(kg[0, 0]), 2)
    13.86

    A flux per year is per pint's year of 365.25 days:

    >>> per_second = xr.DataArray([1.0], dims="x", attrs={"units": "kg m-2 s-1"})
    >>> per_year = convert_units(per_second, "CH4", "kg m-2 yr-1")
    >>> float(per_year[0]) / 86400  # days
    365.25
    """
    if not isinstance(data, xr.Dataset | xr.DataArray):
        raise TypeError("data must be an xarray Dataset or DataArray")

    data = data.copy()
    arrays = [data] if isinstance(data, xr.DataArray) else list(data.data_vars.values())
    plain = all(array.pint.units is None for array in arrays)
    if plain:
        _quantify_from_attrs(arrays)

    # Calculate molecular weight of pollutant
    mw = molecular_weight(pollutant)

    # Use custom pint context to convert mass <--> substance
    with units.context("mass_flux", mw=mw):
        data = data.pint.to(dst_units)
    if plain:
        data = data.pint.dequantify()

    if isinstance(data, xr.Dataset):
        for var in data.data_vars:
            data[var].attrs["units"] = dst_units
    else:
        data.attrs["units"] = dst_units
    return data




def _quantify_from_attrs(arrays: Sequence[DataArray]) -> None:
    """
    Give each array the pint units of its ``units`` attribute, in place.

    Only the values are wrapped. ``.pint.quantify()`` also reads the units of
    the coordinates (``degrees_north``) and drops that attribute from them, which
    a caller who passed plain data does not expect to lose.
    """
    missing = [
        "data" if array.name is None else array.name
        for array in arrays
        if "units" not in array.attrs
    ]
    if missing:
        raise ValueError(
            "Data without pint units needs a 'units' attribute "
            f"(no units attribute on {missing})"
        )
    for array in arrays:
        array.data = units.Quantity(array.data, array.attrs.pop("units"))


def _open_mfdataset(
    files: list[Path], preprocess: Callable[[Dataset], Dataset] | None = None
) -> Dataset:
    """
    Open ``files`` as one dataset, combined the way xarray does it today.

    xarray is changing the defaults of the four arguments named here (it warns
    since 2025.08), and it warns that the new ones change what Vulcan and
    WetCHARTs load and raise on EDGAR v8's annual files. Naming today's values
    keeps the result the same on any xarray, and keeps the warning quiet.

    The files are NetCDF, so ``engine="netcdf4"`` is named too: xarray does not
    then ask each backend whether it can open each file. That saves a few seconds
    when opening the 1,247 files of EDGAR v8's annual methane.
    """
    return xr.open_mfdataset(
        files,
        preprocess=preprocess,
        engine="netcdf4",
        compat="no_conflicts",
        data_vars="all",
        coords="different",
        join="outer",
    )


def _quantify_data_only(
    data: Dataset, units_map: dict[Any, Any] | None = None
) -> Dataset:
    """
    Attach pint units to a dataset's variables while leaving its coordinates alone.

    Real inventory files carry ``units: degrees_north`` on lat/lon, and
    pint-xarray quantifies any variable with a ``units`` attribute unless it is
    told otherwise. A coordinate *index* wrapped in a pint Quantity cannot be
    aligned against a plain one, which silently breaks every operation that
    combines the data with something derived from it -- notably
    :attr:`Inventory.gridcell_area`, and so ``absolute_emissions`` and
    ``integrate``. Mapping each coordinate to ``None`` keeps them plain.
    """
    mapping: dict[Any, Any] = dict(units_map or {})
    mapping.update({coord: None for coord in data.coords})
    return data.pint.quantify(mapping)


def _require_same_units(unit_of: Mapping[str, Any]) -> None:
    """
    Raise a ``ValueError`` unless every sector is in the same units.

    ``unit_of`` maps each sector to its units: a string (the ``units`` attribute,
    or ``None`` when it has none) or a pint unit. Two spellings of one unit
    (``kg/m2/s``, ``kg m-2 s-1``) are the same.
    """
    values = list(unit_of.values())
    if all(u == values[0] for u in values):
        return
    if all(u is not None for u in values) and len({units.Unit(u) for u in values}) == 1:
        return
    listed = ", ".join(
        f"{sector}: {u:~C}" if isinstance(u, pint.Unit) else f"{sector}: {u}"
        for sector, u in unit_of.items()
    )
    raise ValueError(
        f"The sectors are in different units ({listed}). Convert them to one unit "
        "first: Inventory.convert_units, or standardize_units=True when building "
        "the inventory."
    )

def sum_sectors(data: Dataset, exclude: list[str] | None = None) -> DataArray:
    """
    Sum emissions from all sectors in the dataset.

    What "all sectors" covers depends on the product: see each inventory's docstring
    (e.g. EPA v1 annual includes ``Forest_Fires``, EPA v2 express includes the
    supplemental ``PostMeter``).

    Parameters
    ----------
    data : xr.Dataset
        The inventory data with emissions from multiple sectors as different
        variables, each with a ``units`` attribute, all in the same units.
    exclude : list of str, optional
        Sectors to leave out of the sum, e.g. ``["Forest_Fires"]``.

    Returns
    -------
    xr.DataArray
        The sum of emissions from the sectors, in their units.

    Raises
    ------
    ValueError
        If ``exclude`` names a sector that is not in ``data``, or if the sectors
        that are summed are in different units (the numbers cannot be added as
        they are; convert them to one unit first).

    See Also
    --------
    Inventory.total_emissions : This sum of an inventory's data.
    Inventory.integrate : The total over a region, as a mass.

    Examples
    --------
    A sector with no emission in a cell (NaN) adds nothing there:

    >>> import xarray as xr
    >>> from lair.inventories import sum_sectors
    >>> units = {"units": "kg m-2 s-1"}
    >>> data = xr.Dataset(
    ...     {
    ...         "energy": ("x", [1.0, float("nan")], units),
    ...         "waste": ("x", [2.0, 3.0], units),
    ...     }
    ... )
    >>> sum_sectors(data).values
    array([3., 3.])
    >>> sum_sectors(data, exclude=["energy"]).values
    array([2., 3.])
    """
    if exclude:
        unknown = sorted(set(exclude) - set(data.data_vars))
        if unknown:
            raise ValueError(
                f"Not sectors in this inventory: {unknown}. "
                f"Sectors: {sorted(map(str, data.data_vars))}"
            )
        data = data.drop_vars(exclude)
    _require_same_units(
        {str(var): data[var].attrs.get("units") for var in data.data_vars}
    )
    total = data.to_array(dim="sector", name="emissions").sum("sector")
    total.attrs["long_name"] = "Total Emissions"
    total.attrs["units"] = data[list(data.data_vars)[0]].attrs["units"]

    return total




def _label_period_starts(data: Dataset, bounds: str = "time_bnds") -> Dataset:
    """
    Label each time step by the start of its period, read from CF time bounds.

    Some providers label a step by its middle (Vulcan, WetCHARTs); lair labels
    every step by its start (see :class:`Inventory`).

    Parameters
    ----------
    data : xr.Dataset
        Data with a ``time`` coordinate and a decoded ``(time, 2)`` bounds variable.
    bounds : str, optional
        The name of the bounds variable, by default 'time_bnds'.

    Returns
    -------
    xr.Dataset
        The data with ``time`` set to the lower bound of each step.
    """
    start = data[bounds][:, 0]
    if not np.issubdtype(start.dtype, np.datetime64):
        raise ValueError(f"{bounds!r} was not decoded to datetimes ({start.dtype})")
    return data.assign_coords(time=start.values)


def _seconds_per_step(data: Dataset, time_step: str) -> DataArray:
    """
    Return the true calendar length of each time step of ``data``, in seconds.

    Leap years have 366 days and months their actual number of days. lair
    follows the calendar rather than any product's own convention; EDGAR
    (v7 ``cell_method``, v8 ``global_total``) uses the calendar too.
    """
    time = data.time
    if time_step == "annual":
        seconds_per_step = (
            (time.dt.is_leap_year * 366 + (~time.dt.is_leap_year) * 365) * 24 * 3600
        )
    elif time_step == "monthly":
        seconds_per_step = time.dt.days_in_month * 24 * 3600
    elif time_step == "quarterly":
        t = time.to_index()  # days in each 3-month quarter starting at `time`
        seconds_per_step = (
            ((t + pd.DateOffset(months=3)) - t).days.to_numpy() * 24 * 3600
        )
    elif time_step == "biweekly":
        seconds_per_step = np.full(time.size, 14 * 24 * 3600)
    elif time_step == "daily":
        seconds_per_step = np.full(time.size, 24 * 3600)
    elif time_step == "hourly":
        seconds_per_step = np.full(time.size, 3600)
    else:
        raise ValueError(f"Time step {time_step} not supported")
    if isinstance(seconds_per_step, xr.DataArray):
        seconds_per_step = seconds_per_step.data
    return data.assign(sec_per_step=("time", seconds_per_step)).sec_per_step


def _is_per_step(unit: Any, time_step: str) -> bool:
    """
    Whether ``unit`` is a rate per one time step of ``time_step``.

    A flux per year with annual data, per month with monthly data, and so on
    (``_STEP_UNITS``). The unit is read as quantity, area and time, as
    :meth:`Inventory.get_units` reads it.
    """
    step_unit = _STEP_UNITS.get(time_step)
    if step_unit is None or unit is None:
        return False
    parts = f"{units.Unit(unit):~C}".split("/")
    return len(parts) == 3 and units.Unit(parts[2]) == units.Unit(step_unit)


@_keep_attrs
def _convert_units_by_step(
    data: Dataset, pollutant: str, dst_units: Any, time_step: str
) -> Dataset:
    """
    Convert pint-quantified inventory data, with the calendar for a per-step unit.

    A flux per time step (per year with annual data, per month with monthly data)
    covers the calendar length of that step: 366 days in a leap year, the days of
    each month. pint alone would use its year of 365.25 days and a month of 1/12
    of that. So a variable in a per-step unit is first made a flux per second
    with each step's own length, and a per-step ``dst_units`` is reached from a
    flux per second the same way. Any other conversion is :func:`convert_units`.
    """
    dst_per_step = _is_per_step(dst_units, time_step)
    src_per_step = [
        var for var in data.data_vars if _is_per_step(data[var].pint.units, time_step)
    ]
    if "time" not in data.coords or not (dst_per_step or src_per_step):
        return convert_units(data, pollutant, dst_units)

    step = units(_STEP_UNITS[time_step])
    seconds = _seconds_per_step(data, time_step) * units("s")
    data = data.copy()
    for var in src_per_step:
        data[var] = data[var] * step / seconds
    if dst_per_step:
        per_second = units.Unit(dst_units) * step.units / units.Unit("s")
        data = convert_units(data, pollutant, per_second)
        data = data * seconds / step
    return convert_units(data, pollutant, dst_units)

class Inventory(BaseGrid):
    """
    Base class for inventories.

    An inventory holds a flux on a grid, one variable per sector, with pint
    units. The subclasses (:class:`EDGARv8` and the others) read a product's
    files; this class also takes a dataset of your own.

    Time labels mark the **start** of each time step, whatever the provider
    uses: Jan 1 for annual data, the 1st of the month for monthly, 00:00 for
    daily and the top of the hour for hourly. So ``sel(time="2020-01-01")``
    selects January 2020 in every monthly inventory.

    Each parameter but ``data`` and ``standardize_units`` is also an attribute
    of the same name.

    Parameters
    ----------
    data : str | Path | xr.Dataset
        The inventory data. If a string or Path, the path to the data. If an xr.Dataset, the data itself.
        Data should have dimensions of time, lat, lon and each variable should be a different emissions sector.
        If the source data is not in the correct format, the `_process` method should be overridden.
    pollutant : str
        The pollutant, as a chemical formula (``'CH4'``). An all-lowercase name
        is upper-cased (``'ch4'``); a mixed-case one is kept (``'NOx'``).
    src_units : str, optional
        The source units of the data, by default None. If given, every variable
        is taken to be in these units. If None, each variable's own ``units``
        attribute is used (and every variable must have one). Variables in
        different units are kept as they are, but the totals
        (:attr:`total_emissions`, :meth:`integrate`) raise a ``ValueError``
        until they are converted to one unit: ``standardize_units=True``, or
        :meth:`convert_units`.
    time_step : str, optional
        The time step of the data, by default 'annual': ``'annual'``,
        ``'monthly'``, ``'quarterly'``, ``'biweekly'``, ``'daily'`` or
        ``'hourly'``. :attr:`absolute_emissions` multiplies by its length.
    crs : str, optional
        The CRS of the data, by default 'EPSG:4326'.
    version : str, optional
        The version of the inventory, by default None.
    standardize_units : bool, optional
        Whether to standardize the units of all variables to the same units, by default False.
        If True, all variables are assumed to be emissions and are converted to
        :data:`DST_UNITS` (``kg km-2 d-1``) as :meth:`convert_units` converts them.
        If False, the units of each variable are retained as-is.

    Raises
    ------
    FileNotFoundError
        If ``data`` is a path with no files.
    ValueError
        If ``data`` is neither a path nor a Dataset, or a variable has no units.

    See Also
    --------
    EDGARv8 : An inventory read from EDGAR's files.
    sum_sectors : The sum of the sectors.

    Examples
    --------
    An inventory of your own: one sector, a flux of 1e-9 kg m-2 s-1 on four
    cells of 0.1 degree, in 2019 and 2020:

    >>> import numpy as np
    >>> import pandas as pd
    >>> import xarray as xr
    >>> from lair.inventories import Inventory
    >>> ds = xr.Dataset(
    ...     {"energy": (("time", "lat", "lon"), np.full((2, 2, 2), 1e-9))},
    ...     coords={
    ...         "time": pd.date_range("2019-01-01", periods=2, freq="YS"),
    ...         "lat": [40.05, 40.15],
    ...         "lon": [-111.95, -111.85],
    ...     },
    ... )
    >>> inv = Inventory(ds, pollutant="CH4", src_units="kg m-2 s-1")
    >>> inv.get_units()
    ('kg', 'm**2', 's')
    >>> (inv.integrate() / 1e3).values.round()  # tonnes; 2020 has a day more
    array([11943., 11976.])
    """

    def __init__(
        self,
        data: str | Path | Dataset,
        pollutant: str,
        src_units: str | None = None,
        time_step: str = "annual",
        crs: str = "EPSG:4326",
        version: str | None = None,
        standardize_units: bool = False,
    ) -> None:
        """Initialize the inventory; the class docstring describes the parameters."""
        # Upper-case all-lowercase names ('ch4' -> 'CH4') but keep mixed-case
        # formulas as given: 'NOx'.upper() would no longer be a formula
        self.pollutant: str = (
            pollutant if any(c.isupper() for c in pollutant) else pollutant.upper()
        )
        self.time_step: str = time_step
        self.crs = CRS(crs)
        self.version: str | None = version

        self.path: str | None
        if isinstance(data, str | Path):
            self.path = str(data)

            # Open dataset
            files = self.get_files()
            if not files:
                raise FileNotFoundError(
                    f"No {type(self).__name__} files found under {self.path}"
                )
            # Loaders' arithmetic (ensemble means, scale factors, unit
            # factors) must keep the variables' attributes
            with xr.set_options(keep_attrs=True):
                ds = self._open(files)

                # Apply inventory-specific processing
                ds = self._process(ds)
        elif isinstance(data, Dataset):
            self.path = None
            ds = data
        else:
            raise ValueError("Data must be a path to a file or an xarray Dataset")
        if src_units is None:
            # Each variable keeps the units in its own attributes
            var_units = {var: ds[var].attrs.get("units") for var in ds.data_vars}
            missing = [var for var, u in var_units.items() if u is None]
            if missing:
                raise ValueError(
                    "Units must be provided in the data attributes or as an "
                    f"argument (no units attribute on {missing})"
                )
            if len(set(var_units.values())) == 1:
                src_units = next(iter(var_units.values()))

        # Standardize units
        # - Requirements:
        #   - all variables are emissions
        #   - all in the same units
        # src_units is None only when the variables carry different units
        self.src_units: str | pint.registry.Unit | None = src_units
        ds = self._quantify(ds)
        if standardize_units:
            ds = _convert_units_by_step(ds, self.pollutant, DST_UNITS, self.time_step)

        # Set the rioxarray CRS. Projected inventories (e.g. Vulcan) are on x/y
        # with 2D lat/lon coords; the rest are on 1D lat/lon.
        x_dim, y_dim = ("lon", "lat") if "lon" in ds.dims else ("x", "y")

    @property
    @_keep_attrs
    def absolute_emissions(self) -> Dataset:
        """
        The mass emitted in each grid cell in each time step, sector by sector.

        Each sector's flux is multiplied by the area of its grid cell and by the
        length of the time step. The area and time units cancel, so the values
        are in the inventory's own mass (or amount) unit: ``kg`` for a flux in
        ``kg/m**2/s``, ``Mg`` for one in ``Mg/km**2/yr``. With annual data, each
        value is the mass the cell emitted in that year; with monthly data, in
        that month.

        Use it for the mass in one cell, or a sector's total over a region
        (``absolute_emissions.sum(["lat", "lon"])``). For the total of every
        sector over the region, use :meth:`Inventory.integrate`, which is that sum
        added over the sectors. :ref:`The inventory example <inventory-totals>`
        shows the three sums side by side.

        The length of a time step:

        - When the flux is a rate per time step (``a-1`` with annual data,
          ``month-1`` with monthly, ``d-1`` with daily, ``hr-1`` with hourly
          ...), each value is multiplied by exactly one step. This avoids
          pint's Julian year (365.25 d) and month (1/12 of that), so
          1 Mg km-2 a-1 over 1 km2 is exactly 1 Mg in every year.
        - Any other rate (e.g. ``s-1``) is multiplied by the true calendar
          length of each step: 366 days in leap years, the actual days in each
          month. This matches EDGAR, whose leap-year files state
          ``interval: 1 year, 366 days``.

        So the unit of the flux does not matter. :meth:`Inventory.convert_units`
        gives a rate per time step the same calendar length, so converting
        ``kg/m**2/s`` to ``Gg/km**2/yr`` first gives the same masses.

        Returns
        -------
        xr.Dataset
            One variable per sector, on the inventory's grid and time steps, each
            with its mass unit in its ``units`` attribute.

        See Also
        --------
        Inventory.integrate : The total over the grid, every sector added.
        Inventory.total_emissions : The flux in each cell, every sector added.
        """
        # Extract current unit information
        _, area_unit, time_unit = self.get_units(pint=False)

        # Multiply by the gridcell area to get mass|substance per time per time step
        absolute = self._data * (self.gridcell_area * units("km**2")).pint.to(area_unit)

        # A rate per time step covers exactly one step: cancel the time unit
        step_unit = _STEP_UNITS.get(self.time_step)
        if step_unit is not None and units.Unit(time_unit) == units.Unit(step_unit):
            absolute = absolute * units(time_unit)
        else:
            seconds_per_step = _seconds_per_step(self._data, self.time_step)
            # Multiply by the time in the time step to get mass|substance per gridcell
            absolute = absolute * (seconds_per_step * units("s")).pint.to(time_unit)

        absolute.attrs = {
            "long_name": "Absolute Emissions",
            "standard_name": f"{self.time_step.lower()}_emissions_per_gridcell",
        }
        return absolute.pint.dequantify(DEFAULT_PINT_FMT)

    def convert_units(self, dst_units: str, inplace: bool = False) -> Self:
        """
        Convert the units of the data to the desired output units.

        A flux per time step, per year on annual data or per month on monthly
        data (``time_step``), is a flux over the calendar length of that step:
        366 days in a leap year, 365 in any other, and the days of each month.
        pint alone, as in the plain-array :func:`convert_units`, would use its
        year of 365.25 days and a twelfth of it for a month. So the totals,
        :attr:`absolute_emissions` and :meth:`integrate`, are the same in any
        unit. Any other unit (per second, per day on annual data) converts as
        pint converts it. An amount (mol) and a mass (kg) convert through the
        molecular weight of the inventory's pollutant.

        Parameters
        ----------
        dst_units : Any
            The destination units.
        inplace : bool, optional
            Whether to modify the data in place, by default False.

        Returns
        -------
        Inventory
            The inventory with converted units. If `inplace=True` returns `self`,
            otherwise returns a new Inventory copy.

        See Also
        --------
        convert_units : The same for a plain array or dataset, with pint's year.

        Examples
        --------
        A flux of 1 kg m-2 s-1 in 2019 and 2020, per year:

        >>> import numpy as np
        >>> import pandas as pd
        >>> import xarray as xr
        >>> from lair.inventories import Inventory
        >>> ds = xr.Dataset(
        ...     {"energy": (("time", "lat", "lon"), np.ones((2, 1, 1)))},
        ...     coords={
        ...         "time": pd.date_range("2019-01-01", periods=2, freq="YS"),
        ...         "lat": [40.05],
        ...         "lon": [-111.95],
        ...     },
        ... )
        >>> inv = Inventory(ds, pollutant="CH4", src_units="kg m-2 s-1")
        >>> per_year = inv.convert_units("kg m-2 yr-1")
        >>> per_year.data["energy"].values.ravel() / 86400  # days in each year
        array([365., 366.])
        """
        data = _convert_units_by_step(
            self._data, self.pollutant, dst_units, self.time_step
        )
        if inplace:
            self._data = data
            self.src_units = dst_units
            return self
        else:
            new = self.copy()
            new._data = data
            new.src_units = dst_units
            return new

    @_keep_attrs
    def integrate(self) -> DataArray:
        """
        Return the total mass emitted over the whole grid in each time step.

        This is the regional total: :attr:`Inventory.absolute_emissions` (each
        cell's flux times its area and the length of the time step) summed over
        the cells and over the sectors, so ``clip()`` the inventory to the region
        first. The result is in the inventory's own mass (or amount) unit,
        ``kg`` for a flux in ``kg/m**2/s``, and each value is the mass emitted
        in one time step: kilograms per year with annual data, per month with
        monthly data.

        Unlike :attr:`Inventory.total_emissions`, which sums only the sectors
        and is still a flux per area, this is a mass. :ref:`The inventory example
        <inventory-totals>` shows both, with a check by hand.

        Returns
        -------
        xr.DataArray
            The total, along ``time`` only, with the mass unit in its ``units``
            attribute. It is lazy for an inventory read from files:
            ``compute()`` reads them.

        See Also
        --------
        Inventory.absolute_emissions : The mass in each cell, by sector.
        Inventory.total_emissions : The flux in each cell, every sector added.
        """
        x_dim, y_dim = self.data.rio.x_dim, self.data.rio.y_dim
        cells = self.absolute_emissions.sum([x_dim, y_dim])
        # Summed over the cells, the per-gridcell names no longer describe it
        cells.attrs = {}
        return sum_sectors(cells)

    """
    Base class for inventories that are multi-model.

    The files hold an ensemble along a ``model`` dimension. The inventory keeps
    one member, or the mean or median of them, and the whole ensemble in
    ``multimodel_data``.

    Parameters
    ----------
    data : str | Path | xr.Dataset
        As for :class:`Inventory`, with a ``model`` dimension.
    pollutant : str
        As for :class:`Inventory`.

        As for :class:`Inventory`.
    time_step : str, optional
        As for :class:`Inventory`.
    crs : str, optional
        As for :class:`Inventory`.
    version : str, optional
        As for :class:`Inventory`.
    model : int | str, optional
        The model to select from the multimodel data, by default None.
        Either a value of the ``model`` coordinate (a numeric string is cast
        to match an integer coordinate), ``'mean'`` or ``'median'`` for the
        ensemble mean or median. If None, the mean of all models is used.

    See Also
    --------
    WetCHARTs : The multi-model inventory lair reads.
    """

    multimodel_data: Dataset

    def __init__(
        self,
        data: str | Path | Dataset,
        pollutant: str,
        src_units: str | None = None,
        time_step: str = "annual",
        crs: str = "EPSG:4326",
        version: str | None = None,
        model: int | str | None = None,
    ) -> None:
        """Initialize the inventory; the class docstring describes the parameters."""
        self.model = model or "mean"
        super().__init__(
            data,
            pollutant,
            src_units=src_units,
            time_step=time_step,
            crs=crs,
            version=version,
        )

    def _process(self, data: Dataset) -> Dataset:
        self.multimodel_data = data

        if self.model == "mean":
            return data.mean(dim="model")
        elif self.model == "median":
            return data.median(dim="model")
        model = self.model
        if np.issubdtype(data["model"].dtype, np.integer):
            model = int(model)  # e.g. WetCHARTs codes: '1913' -> 1913
        return data.sel(model=model)




#: The year an EDGAR file covers: four digits between underscores in its name
#: (``v8.0_FT2022_GHG_CH4_2020_AGS_flx.nc``). The release tag (``FT2022``) has
#: letters before its digits, so it does not match.
_EDGAR_YEAR = re.compile(r"_(\d{4})_")


def _year_spans(years: Sequence[int]) -> str:
    """
    Write sorted ``years`` as spans: ``[1970, 1971, 1972, 2020]`` is ``'1970 to 1972, 2020'``.
    """
    spans = []
    # Years in a row share the difference between their value and their position
    for _, group in itertools.groupby(
        enumerate(years), key=lambda pair: pair[1] - pair[0]
    ):
        run = [year for _, year in group]
        first, last = run[0], run[-1]
        spans.append(str(first) if first == last else f"{first} to {last}")
    return ", ".join(spans)

    EDGAR - Emissions Database for Global Atmospheric Research

    https://edgar.jrc.ec.europa.eu/

    EDGAR is a multipurpose, independent, global database of anthropogenic
    emissions of greenhouse gases and air pollution on Earth. EDGAR provides
    independent emission estimates compared to what reported by European Member
    States or by Parties under the United Nations Framework Convention on Climate
    Change (UNFCCC), using international statistics and a consistent IPCC methodology.

    EDGAR provides both emissions as national totals and gridmaps at 0.1 x 0.1 degree
    resolution at global level, with yearly, monthly and up to hourly data.

    Fluxes are annual or monthly means in kg m-2 s-1, so totals depend on the
    year length. lair uses the true calendar (366 days in leap years), which is
    also EDGAR's: v7 files state ``cell_method = "time: mean (interval: 1 year,

    the year is in the file's name. ``start`` and ``end``, the first and last year

    def get_files(self) -> list[Path]:
        """
        Recursively get the inventory files for each sector.

        If the inventory was made with ``start`` or ``end``, only the files of
        those years are returned, chosen by the year in each file's name: no
        file is opened.

        Returns
        -------
        list[Path]
            The inventory files.

        Raises
        ------
        FileNotFoundError
            If there are files, but none for the years asked.

        See Also
        --------
        Inventory.get_files : The files of an inventory that is not EDGAR.
        """
        files = [f for f in self._file_root().rglob("*.nc") if "TOTALS" not in f.stem]
        if not files or (self._start is None and self._end is None):
            return files  # nothing to choose from (Inventory.__init__ raises) or all

        year_of = {
            f: int(m.group(1)) for f in files if (m := _EDGAR_YEAR.search(f.name))
        }
        chosen = [
            f
            for f, year in year_of.items()
            if (self._start is None or year >= self._start)
            and (self._end is None or year <= self._end)
        ]
        name = type(self).__name__
        if not chosen:
            has = _year_spans(sorted(set(year_of.values()))) or "none"
            raise FileNotFoundError(
                f"No {name} files for start={self._start}, end={self._end} under "
                f"{self.path} (years with files: {has})"
            )
        found = {year_of[f] for f in chosen}
        wanted = range(
            self._start if self._start is not None else min(found),
            (self._end if self._end is not None else max(found)) + 1,
        )
        missing = [year for year in wanted if year not in found]
        if missing:
            logger.warning(
                "%s has no files for %s under %s",
                name,
                _year_spans(missing),
                self.path,
            )
        return chosen

    def get_sector_name(self, sector: str) -> str:
        """
        Generate a formatted name from the sector description.

        Parameters
        ----------
        sector : str
            The key of the sector in the sectors dictionary.

        Returns
        -------
        str
            The formatted name. Sectors missing from ``sectors`` keep their code.

        See Also
        --------
        EDGARv7 : Names its variables with it.

        Examples
        --------
        The variable that EDGAR v7's ``ENF`` files become, for an ``EDGARv7``

            inv.get_sector_name("ENF")  # 'Enteric_fermentation'

        return name

    GHG Database (a collaboration between the European Commission, Joint
    Research Centre (JRC), the International Energy Agency (IEA), and
    comprising IEA-EDGAR CO2, EDGAR CH4, EDGAR N2O, EDGAR F-GASES version 7.0,

        Root of the inventory archive, by default ``$LAIR_INVENTORY_DIR``.

        The first year to read, by default the first year the archive has.
    end : int, optional
        The last year to read (included), by default the last year the archive

    Raises
    ------
    FileNotFoundError
        If there are no files for the years from ``start`` to ``end``.

        If ``start`` or ``end`` is not an integer year.

    Notes
    -----
    The folder is the pollutant's name as typed (``'CH4'``), below the
    inventory root (``inventory_dir``, default ``$LAIR_INVENTORY_DIR``), with
    one folder per sector. lair takes the year from each file name and does not
    read the ``TOTALS`` folder::

        EDGAR/v7/CH4/
          AGS/v7.0_FT2021_CH4_2021_AGS.0.1x0.1.nc
          AWB/v7.0_FT2021_CH4_2021_AWB.0.1x0.1.nc

    >>> from lair.inventories import EDGARv7
    >>> inv = EDGARv7("CH4", start=2021)  # doctest: +SKIP
    >>> "Enteric_fermentation" in inv.data  # doctest: +SKIP
    True

    def __init__(
        self,
        pollutant: str,
        inventory_dir: str | None = None,
        start: int | None = None,
        end: int | None = None,
    ) -> None:
        """Initialize the EDGAR inventory; the class docstring describes the parameters."""
        self._set_years(start, end)
        self.edgar_dir = os.path.join(
            get_data_dir(INVENTORY_DIR_ENV, inventory_dir), "EDGAR"
        )
        path = os.path.join(self.edgar_dir, self.version, pollutant)
        super().__init__(
            path, pollutant, src_units=self.src_units, version=self.version
        )

    def _preprocess(self, ds: Dataset) -> Dataset | None:
        # Need to strip year and sector name from filename
        filename = ds.encoding["source"]

        pattern = r"_(\d{4})_([\w_]+)\.0\.1x0\.1\.nc$"
        match = re.search(pattern, filename)

        if match:
            # Extract the year and the text between the year and flx.nc
            year = match.group(1)
            sector_code = match.group(2)
        else:
            return None

        var = self.get_sector_name(sector_code)

        ds = ds.rename({f"emi_{self.pollutant.lower()}": var})
        ds[var].attrs["long_name"] = f"{var}_Emissions"
        ds[var].attrs["standard_name"] = self.get_standard_name()

        # Add time coordinate
        ds = ds.expand_dims(time=[dt.datetime(int(year), 1, 1)])
        return ds

    def _process(self, data: Dataset) -> Dataset:
        data = data.assign_coords(lon=wrap_lons(data.lon)).sortby("lon")

        # Round grid to nearest 0.1 degrees
        # - cell coordinates are center-of-cell, so we actually need to round to nearest 0.05
        data = round_latlon(data, 2, 2)
        return data

    EDGAR v8 - Global Greenhouse Gas Emissions

    https://edgar.jrc.ec.europa.eu/dataset_ghg80

    EDGAR (Emissions Database for Global Atmospheric Research) Community
    GHG Database, a collaboration between the European Commission, Joint

    0.15 Mt gap is an inconsistency within EDGAR's own files, not data lair
    fails to read; EDGAR's ``TOTALS`` follow ``PRO_FFF``, so the sum of the
    sectors is 0.15 Mt below them (see GitHub issue #37).

    Parameters
    ----------
    pollutant : str
        The pollutant, as the name of its folder (``'CH4'``).
    time_step : Literal['annual', 'monthly'], optional
        The time step of the data, by default 'annual'.
    inventory_dir : str, optional
        Root of the inventory archive, by default ``$LAIR_INVENTORY_DIR``.
    start : int, optional
        The first year to read, by default the first year the archive has.
    end : int, optional
        The last year to read (included), by default the last year the archive
        has. Only the files of the years from ``start`` to ``end`` are opened,
        as :class:`EDGAR` says.

    Raises
    ------
    FileNotFoundError
        If there are no files for the years from ``start`` to ``end``.
    TypeError
        If ``start`` or ``end`` is not an integer year.

    Notes
    -----
    The annual files are in ``EDGAR/v8/CH4/``, as :ref:`the inventory example
    <inventory-data>` shows. The monthly files (``time_step="monthly"``) are in
    ``EDGAR/v8/monthly/CH4/``: one file per sector group and year, with the
    groups ``AGRICULTURE``, ``BUILDINGS``, ``FUEL_EXPLOITATION``,
    ``IND_COMBUSTION``, ``IND_PROCESSES``, ``POWER_INDUSTRY``, ``TRANSPORT`` and
    ``WASTE`` each in a folder of its own::

        EDGAR/v8/monthly/CH4/
          AGRICULTURE/v8.0_FT2022_GHG_CH4_2000_AGRICULTURE_flx.nc
          AGRICULTURE/v8.0_FT2022_GHG_CH4_2001_AGRICULTURE_flx.nc
          ...
          WASTE/v8.0_FT2022_GHG_CH4_2022_WASTE_flx.nc

    Without arguments every year is read. ``start`` and ``end`` open only the
    files of those years (both included); :ref:`the inventory example
    <inventory-load>` says how long every year takes.

    See Also
    --------
    EDGARv7 : The version before.
    Inventory.integrate : The total over a region.

    Examples
    --------
    The methane EDGAR v8 puts in the Salt Lake Valley each year, in kg, as
    :doc:`/examples/inventory` computes it. It reads ``$LAIR_INVENTORY_DIR``, so
    these lines are not run when the docs build:

    >>> from lair.inventories import EDGARv8
    >>> inv = EDGARv8("CH4", start=2020, end=2022)  # doctest: +SKIP

    >>> slv.integrate().compute()  # doctest: +SKIP

    version: str = "v8"

    def __init__(
        self,
        pollutant: str,
        time_step: Literal["annual", "monthly"] = "annual",
        inventory_dir: str | None = None,
        start: int | None = None,
        end: int | None = None,
    ):
        """Initialize the EDGAR inventory; the class docstring describes the parameters."""
        self._set_years(start, end)
        self.edgar_dir = os.path.join(
            get_data_dir(INVENTORY_DIR_ENV, inventory_dir), "EDGAR"
        )
        path = os.path.join(
            self.edgar_dir,
            self.version,
            "" if time_step == "annual" else time_step,
            pollutant,
        )
        super().__init__(
            path,
            pollutant,
            src_units=self.src_units,
            time_step=time_step,
            version=self.version,
        )

    def _preprocess(self, ds: Dataset) -> Dataset:
        # Rename var and add attributes
        old_var = "fluxes"
        var = (
            ds[old_var]
            .attrs["long_name"]
            .replace(" ", "_")
            .replace(",", "")
            .replace("-", "_")
        )
        ds = ds.rename({old_var: var})
        attrs = ds[var].attrs
        attrs["long_name"] = f"{var}_Emissions"
        attrs["standard_name"] = self.get_standard_name()

        if self.time_step == "annual":
            # Add time coordinate to annual data
            ds = ds.expand_dims(time=[dt.datetime(int(attrs["year"]), 1, 1)])
        else:
            # Monthly files label each month by its 15th and have no time
            # bounds: label by the 1st instead
            time = pd.DatetimeIndex(ds.indexes["time"])
            ds = ds.assign_coords(time=time.to_period("M").to_timestamp())
        return ds

    def _process(self, data: Dataset) -> Dataset:
        # Annual files have Fuel_exploitation (PRO_FFF) alongside its COAL, GAS
        # and OIL components; drop it so the sectors don't double count. The
        # components fall slightly short of it (CH4 2020: 109.27 vs 109.42 Mt),
        # an inconsistency in EDGAR's files (see the class docstring). Monthly
        # files only have Fuel_exploitation: keep it there.
        components = [f"Fuel_exploitation_{f}" for f in ("COAL", "GAS", "OIL")]
        if all(var in data.data_vars for var in components):
            data = data.drop_vars(["Fuel_exploitation"], errors="ignore")
        return data

class EPA(Inventory, metaclass=ABCMeta):
    """
    United States Environmental Protection Agency (EPA) Gridded Methane Emissions

    https://www.epa.gov/ghgemissions/us-gridded-methane-emissions

    The gridded EPA U.S. methane greenhouse gas inventory (gridded methane GHGI)
    includes time series of annual methane emission maps with 0.1° x 0.1°
    (~ 10 x 10 km) spatial and monthly temporal resolution for the contiguous
    United State (CONUS). This gridded methane inventory is designed to be
    consistent with methane emissions from the U.S. EPA Inventory of U.S.
    Greenhouse Gas Emissions and Sinks (U.S. GHGI).

    lair's totals differ a little from EPA's own. EPA's files give each cell's
    area on a sphere of 6371 km (their ``grid_cell_area``, which lair drops);
    lair uses the area on the WGS84 ellipsoid, as for every inventory
    (:func:`lair.geo.gridcell_area`). A 0.1 degree cell in the Salt Lake Valley
    (40.75 N) is 93.783 km2 on the ellipsoid and 93.668 km2 on the sphere, so
    lair's totals there are about 0.12% above EPA's. The difference changes
    with latitude: about 0.2% below at 25 N and 0.3% above at 49 N.

    Parameters
    ----------
    data : str | Path
        The file or folder, which :class:`EPAv1` and :class:`EPAv2` build from
        their own arguments.
    pollutant : str
        ``'CH4'``.
    src_units : str
        ``'molec cm-2 s-1'``, EPA's units.
    time_step : str
        As :class:`EPAv1` and :class:`EPAv2` read it.
    crs : str
        ``'EPSG:4326'``.
    version : str
        ``'v1'`` or ``'v2'``.
    standardize_units : bool
        False.

    See Also
    --------
    EPAv2 : The gridded GHGI, 2012 to 2020.
    EPAv1 : The 2012 inventory.
    """

    pollutant: str = "CH4"
    src_units: str = "molec cm-2 s-1"

    _emissions_prefix: str
    _variable_pattern: str = r"{}(?:_Supp)?_([1-9][A-Z]?[1-9]?[a-z]*)_([A-Za-z_]*)"
    _lat_deci = 2
    _lon_deci = 2

    def _extract_ipcc_code_and_short_name(self, var_name, prefix):
        pattern = self._variable_pattern.format(prefix)
        match = re.search(pattern, var_name)
        if match:
            ipcc_code = match.group(1)
            short_name = match.group(2)
            return ipcc_code, short_name
        else:
            return None, None

    def _process(self, data: Dataset) -> Dataset:
        # Drop grid_cell_area variable, use gridcell_area instead
        data = data.drop_vars("grid_cell_area", errors="ignore")

        # Rename variables and add attributes
        name_dict = {}
        for var in data.data_vars:
            attrs = data[var].attrs
            if str(var).startswith(self._emissions_prefix):
                ipcc_code, short_name = self._extract_ipcc_code_and_short_name(
                    var, self._emissions_prefix
                )
                attrs["IPCC_Code"] = ipcc_code
                attrs["long_name"] = f"{short_name}_Emissions"
                attrs["standard_name"] = self.get_standard_name()
                attrs.pop(
                    "source_category", None
                )  # remove source_category attribute for v2 (same as IPCC code)
                name_dict[var] = short_name
        data = data.rename(name_dict)

        # Round grid to nearest 0.1 degrees
        # - cell coordinates are center-of-cell, so we actually need to round to nearest 0.05
        data = round_latlon(data, self._lat_deci, self._lon_deci)
        return data

    def get_monthly_scale_factors(self) -> Dataset:
        """
        Get the monthly scale factors.

        Returns
        -------
        xr.Dataset
            The monthly scale factors, one variable per sector that has them,
            from the files in ``EPA/v2/monthly_scale_factors/`` (2012 to 2018).

        See Also
        --------
        EPAv2 : ``scale_by_month=True`` applies them.
        """
        files = list(
            Path(self.epa_dir, self.version, "monthly_scale_factors").glob("*.nc")
        )
        ds = _open_mfdataset(files)
        ds = ds.rename_vars(
            {var: "_".join(str(var).split("_")[4:]) for var in list(ds.data_vars)}
        )
        return ds



    @staticmethod
    def _apply_monthly_scale_factors(data: Dataset, sf: Dataset) -> Dataset:
        # Expand every sector to the months of `sf` at its annual rate (repeat
        # each year's Jan value), then scale the sectors that have monthly scale
        # factors. Only some sectors have them (petroleum/gas systems, stationary
        # combustion, manure, rice, field burning); the rest (enteric
        # fermentation, landfills, coal, ...) keep their annual rate.
        monthly = data.reindex(time=sf["time"], method="ffill")
        scaled = monthly[list(sf.data_vars)]
        scaled *= sf
        return monthly.assign(scaled.data_vars)

    def _scale_by_month(self, data: Dataset) -> Dataset:
        self.time_step = "monthly"
        sf = self.get_monthly_scale_factors()

        # Round grid coordinates due to floating point errors
        sf = round_latlon(sf, self._lat_deci, self._lon_deci)

        monthly = self._apply_monthly_scale_factors(data, sf)

        if self.express:
            # Only scale _express_vars_scalable_past_2018 past 2018:
            # "For other sources, monthly variability is too year-specific
            # and should not be extrapolated to the express extension dataset
            # for years after 2018"
            sf_2018 = sf[self._express_vars_scalable_past_2018].sel(time="2018")
            express_sf = xr.concat(
                [
                    sf_2018.assign_coords(
                        time=np.array(
                            [dt.datetime(year, month, 1) for month in range(1, 13)]
                        )
                    )
                    for year in range(2019, int(data.time.dt.year.max()) + 1)
                ],
                dim="time",
            )

            # Every other sector keeps its annual rate in every month after 2018
            express_monthly = self._apply_monthly_scale_factors(data, express_sf)
            monthly = xr.concat([monthly, express_monthly], dim="time")

        return monthly

    def _process(self, data: Dataset) -> Dataset:
        data = super()._process(data)

        if self.scale_by_month:
            data = self._scale_by_month(data)

        return data

        filename = ds.encoding["source"]

        var = self._strip_var_names(ds)
        ds = ds.rename_vars({"emis_ch4": var})
        ds[var].attrs = {
            "long_name": f"{var}_Emissions",
            "standard_name": self.get_standard_name(),

        # the units for lat and lon have spaces in them
        # I feel like this is probably a bug in pint-xarray
        data.lon.attrs["units"] = "degrees_east"
        data.lat.attrs["units"] = "degrees_north"

        # Round grid to nearest 0.1 degrees
        # - cell coordinates are center-of-cell, so we actually need to round to nearest 0.05
        data = round_latlon(data, 2, 2)
        return data

        resolution : float | tuple[x_res, y_res]
            The new resolution in degrees. If a single value is provided, the resolution
            is assumed to be the same in both dimensions.
        regrid_method : str, optional
            The regridding method, by default 'conservative'.
        inplace : bool, optional
            Whether to modify the object in place. Default is False.
        force : bool
            Whether to override the clipping requirement

        Returns
        -------
        Vulcan
            The reprojected inventory (on EPSG:4326 lat/lon).

        Raises
        ------
        ValueError
            If the data was not clipped first and ``force`` is False.

        See Also
        --------
        Vulcan.clip : Clip first.
        """
        if not self._is_clipped and not force:
            raise ValueError(
                "Data must be clipped before reprojecting! Set force=True to override"
            )
        return super().reproject(resolution, regrid_method, inplace=inplace)



    def _preprocess(self, ds: Dataset) -> Dataset:
        # Rename variables
        # Annual: Vulcan_v3_US_annual_1km_elec_prod_mn.nc4
        # Hourly: Vulcan.v3.US.hourly.1km.elec_prod.mn.2015.d001.nc4
        # The sector sits between the resolution and the uncertainty code. It
        # can contain underscores, so rejoin the annual parts (which are
        # separated by underscores too)
        filename = os.path.basename(ds.encoding["source"])
        parts = filename.split(self._sep)
        sector = "_".join(parts[5:-1]) if self._sep == "_" else parts[5]
        ds = ds.rename({"carbon_emissions": sector})
        # Times are mid-year / mid-hour: label by the start of each step
        ds = _label_period_starts(ds)
        # Drop unnecessary variables and dims
        ds = ds.drop_vars(["time_bnds", "crs"])
        return ds

    def _open(self, files: list[Path]) -> Dataset:
        return _open_mfdataset(files, preprocess=self._preprocess)

    def _process(self, data: Dataset) -> Dataset:
        # Vulcan marks cells without emissions as NaN (most cells of the point
        # source sectors). Treat them as zero: otherwise every regridded cell
        # touching a NaN becomes NaN and conservative regridding drops most of
        # the airport/cement/cmv/elec_prod emissions.
        data = data.fillna(0)

        # The files are mass of carbon (tC); express them as mass of CO2 to
        # match pollutant='CO2' (see the class note)
        c_to_co2 = float((molecular_weight("CO2") / molecular_weight("C")).magnitude)
        data = data * c_to_co2
        for var in data.data_vars:
            data[var].attrs["comment"] = (
                f"Converted by lair from tonnes of carbon to "
                f"tonnes of CO2 (x {c_to_co2:.4f})."
            )
        return data

class WetCHARTs(MultiModelInventory):
    """
    WetCHARTs - Wetland Methane Emissions and Uncertainty

    https://daac.ornl.gov/CMS/guides/MonthlyWetland_CH4_WetCHARTs.html

    This dataset provides global monthly wetland methane (CH4) emissions
    estimates at 0.5 by 0.5-degree resolution for the period 2001-2019
    that were derived from an ensemble of multiple terrestrial biosphere
    models, wetland extent scenarios, and CH4:C temperature dependencies
    that encompass the main sources of uncertainty in wetland CH4 emissions.
    There are 18 model configurations. WetCHARTs v1.3.1 is an updated product
    of WetCHARTs v1.0 Extended Ensemble.

    Bloom, A.A., K.W. Bowman, M. Lee, A.J. Turner, R. Schroeder, J.R. Worden,
    R.J. Weidner, K.C. McDonald, and D.J. Jacob. 2021. CMS: Global 0.5-deg
    Wetland Methane Emissions and Uncertainty (WetCHARTs v1.3.1). ORNL DAAC,
    Oak Ridge, Tennessee, USA. https://doi.org/10.3334/ORNLDAAC/1915

    Notes
    -----
    One file per year, in ``WetCHARTs/v1.3.1/`` below the inventory root
    (``inventory_dir``, default ``$LAIR_INVENTORY_DIR``). lair reads every
    ``.nc`` file of the folder::

        WetCHARTs/v1.3.1/
          WetCHARTs_v1_3_1_2001.nc
          ...
          WetCHARTs_v1_3_1_2019.nc

    Parameters
    ----------
    model : int | str | None, optional
        The model to select, by default None: a four-digit ensemble member
        code (e.g. ``1913`` or ``'1913'``), ``'mean'`` or ``'median'``.
        If None, the ensemble mean is used.
    inventory_dir : str, optional
        Root of the inventory archive, by default ``$LAIR_INVENTORY_DIR``.

    See Also
    --------
    MultiModelInventory : How the ensemble member is chosen.

    Examples
    --------
    The ensemble median, and every member in ``multimodel_data``. It reads
    ``$LAIR_INVENTORY_DIR``, so these lines are not run when the docs build:

    >>> from lair.inventories import WetCHARTs
    >>> wet = WetCHARTs(model="median")  # doctest: +SKIP
    >>> members = wet.multimodel_data.model.values  # doctest: +SKIP
    """

    version: str = "v1.3.1"
    pollutant = "CH4"
    src_units: str = "mg m-2 d-1"
    time_step = "monthly"

    def __init__(
        self, model: int | str | None = None, inventory_dir: str | None = None
    ):
        """Initialize the inventory; the class docstring describes the parameters."""
        self.wetcharts_dir = os.path.join(
            get_data_dir(INVENTORY_DIR_ENV, inventory_dir), "WetCHARTs"
        )
        path = os.path.join(self.wetcharts_dir, self.version)
        super().__init__(
            path,
            self.pollutant,
            src_units=self.src_units,
            time_step=self.time_step,
            version=self.version,
            model=model,
        )

    def _process(self, data: Dataset) -> Dataset:
        # Times are mid-month: label by the first of the month
        data = _label_period_starts(data)

        # Drop unnecessary variables and dims
        data = data.drop_vars(["time_bnds", "crs"])

        # Rename variables
        data = data.rename({"wetland_CH4_emissions": "wetlands"})
        data.wetlands.attrs["long_name"] = "Wetland_CH4_Emissions"
        data.wetlands.attrs["standard_name"] = self.get_standard_name()

        # Cells outside wetlands are NaN (most of the grid). Treat them as zero,
        # as for Vulcan: otherwise conservative regridding turns every coarse
        # cell touching one into NaN and drops its emissions. Filling before the
        # ensemble statistic also counts a model's non-wetland cell as zero.
        data = data.fillna(0)

        return super()._process(data)  # select a single model
